from flask import Flask, render_template, request, jsonify
from googletrans import Translator, LANGUAGES
from gtts import gTTS
import base64
import io

app = Flask(__name__)
translator = Translator()

@app.route('/')
def home():
    # Get all available languages
    languages = {code: name.capitalize() for code, name in LANGUAGES.items()}
    return render_template('index.html', languages=languages)

@app.route('/translate', methods=['POST'])
def translate():
    try:
        data = request.get_json()
        text = data.get('text', '')
        source_lang = data.get('source', 'auto')
        target_lang = data.get('target', 'en')
        
        if not text.strip():
            return jsonify({'error': 'Please enter text to translate'}), 400
        
        # Perform translation
        if source_lang == 'auto':
            result = translator.translate(text, dest=target_lang)
        else:
            result = translator.translate(text, src=source_lang, dest=target_lang)
        
        return jsonify({
            'translated_text': result.text,
            'detected_language': LANGUAGES.get(result.src, result.src).capitalize(),
            'source_language': result.src
        })
    except Exception as e:
        return jsonify({'error': str(e)}), 500

@app.route('/text-to-speech', methods=['POST'])
def text_to_speech():
    try:
        data = request.get_json()
        text = data.get('text', '')
        lang = data.get('lang', 'en')
        
        if not text.strip():
            return jsonify({'error': 'No text provided'}), 400
        
        # Generate speech
        tts = gTTS(text=text, lang=lang)
        audio_buffer = io.BytesIO()
        tts.write_to_fp(audio_buffer)
        audio_buffer.seek(0)
        
        # Encode to base64 for frontend playback
        audio_base64 = base64.b64encode(audio_buffer.read()).decode('utf-8')
        
        return jsonify({'audio': audio_base64})
    except Exception as e:
        return jsonify({'error': str(e)}), 500

if __name__ == '__main__':
    app.run(debug=True, port=5000)

        let detectedSourceLang = 'en';
        
        async function translateText() {
            const text = document.getElementById('sourceText').value;
            const sourceLang = document.getElementById('sourceLang').value;
            const targetLang = document.getElementById('targetLang').value;
            
            if (!text.trim()) {
                alert('Please enter text to translate');
                return;
            }
            
            document.getElementById('translateBtn').textContent = 'Translating...';
            document.getElementById('translateBtn').disabled = true;
            
            try {
                const response = await fetch('/translate', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ text, source: sourceLang, target: targetLang })
                });
                
                const data = await response.json();
                
                if (data.error) {
                    alert(data.error);
                } else {
                    document.getElementById('translatedText').value = data.translated_text;
                    document.getElementById('detectedLang').textContent = 
                        `Detected Language: ${data.detected_language}`;
                    detectedSourceLang = data.source_language;
                }
            } catch (error) {
                alert('Translation failed. Please try again.');
            }
            
            document.getElementById('translateBtn').textContent = 'Translate';
            document.getElementById('translateBtn').disabled = false;
        }
        
        async function speakText(type) {
            const text = type === 'source' 
                ? document.getElementById('sourceText').value 
                : document.getElementById('translatedText').value;
            const lang = type === 'source' 
                ? (document.getElementById('sourceLang').value === 'auto' ? detectedSourceLang : document.getElementById('sourceLang').value)
                : document.getElementById('targetLang').value;
            
            if (!text.trim()) return;
            
            try {
                const response = await fetch('/text-to-speech', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ text, lang })
                });
                
                const data = await response.json();
                if (data.audio) {
                    const audio = new Audio('data:audio/mp3;base64,' + data.audio);
                    audio.play();
                }
            } catch (error) {
                console.error('TTS failed:', error);
            }
        }
        
        function copyText() {
            const text = document.getElementById('translatedText').value;
            navigator.clipboard.writeText(text).then(() => {
                alert('Copied to clipboard!');
            });
        }
        
        function swapLanguages() {
            const sourceLang = document.getElementById('sourceLang');
            const targetLang = document.getElementById('targetLang');
            const sourceText = document.getElementById('sourceText');
            const translatedText = document.getElementById('translatedText');
            
            if (sourceLang.value !== 'auto') {
                const tempLang = sourceLang.value;
                sourceLang.value = targetLang.value;
                targetLang.value = tempLang;
            }
            
            const tempText = sourceText.value;
            sourceText.value = translatedText.value;
            translatedText.value = tempText;
        }
        
        // Auto-translate on Enter key
        document.getElementById('sourceText').addEventListener('keydown', (e) => {
            if (e.key === 'Enter' && e.ctrlKey) {
                translateText();
            }
        });

3. chatbot.py (NLP Chatbot Engine)

import json
import nltk
from nltk.corpus import stopwords
from nltk.tokenize import word_tokenize
from nltk.stem import WordNetLemmatizer
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import numpy as np
import re

# Download required NLTK data
nltk.download('punkt', quiet=True)
nltk.download('stopwords', quiet=True)
nltk.download('wordnet', quiet=True)
nltk.download('punkt_tab', quiet=True)

class FAQChatbot:
    def __init__(self, faq_file='faqs.json'):
        self.lemmatizer = WordNetLemmatizer()
        self.stop_words = set(stopwords.words('english'))
        self.load_faqs(faq_file)
        self.build_vectorizer()
        
    def load_faqs(self, faq_file):
        """Load FAQ data from JSON file"""
        with open(faq_file, 'r') as f:
            data = json.load(f)
        self.faqs = data['faqs']
        self.questions = [faq['question'] for faq in self.faqs]
        self.answers = [faq['answer'] for faq in self.faqs]
        self.keywords = [faq.get('keywords', []) for faq in self.faqs]
        
    def preprocess_text(self, text):
        """Clean and preprocess text using NLP techniques"""
        # Convert to lowercase
        text = text.lower()
        # Remove special characters and digits
        text = re.sub(r'[^a-zA-Z\s]', '', text)
        # Tokenize
        tokens = word_tokenize(text)
        # Remove stopwords and lemmatize
        processed_tokens = [
            self.lemmatizer.lemmatize(token) 
            for token in tokens 
            if token not in self.stop_words and len(token) > 2
        ]
        return ' '.join(processed_tokens)
    
    def build_vectorizer(self):
        """Build TF-IDF vectorizer from FAQ questions"""
        # Preprocess all questions
        processed_questions = [self.preprocess_text(q) for q in self.questions]
        
        # Also include keywords in the vectorization
        combined_texts = []
        for i, q in enumerate(processed_questions):
            keywords_text = ' '.join(self.keywords[i]) if self.keywords[i] else ''
            combined_texts.append(f"{q} {keywords_text}")
        
        # Build TF-IDF vectorizer
        self.vectorizer = TfidfVectorizer()
        self.faq_vectors = self.vectorizer.fit_transform(combined_texts)
        
    def find_best_match(self, user_question):
        """Find the most similar FAQ to the user's question"""
        # Preprocess user question
        processed_question = self.preprocess_text(user_question)
        
        if not processed_question.strip():
            return None, 0.0
        
        # Vectorize user question
        user_vector = self.vectorizer.transform([processed_question])
        
        # Calculate cosine similarity with all FAQs
        similarities = cosine_similarity(user_vector, self.faq_vectors)[0]
        
        # Find the best match
        best_idx = np.argmax(similarities)
        best_score = similarities[best_idx]
        
        return best_idx, best_score
    
    def get_response(self, user_question):
        """Generate chatbot response for user question"""
        # Handle greetings
        greetings = ['hi', 'hello', 'hey', 'good morning', 'good afternoon', 'good evening']
        if user_question.lower().strip() in greetings:
            return {
                'answer': "Hello! 👋 I'm your FAQ assistant. How can I help you today?",
                'confidence': 1.0,
                'matched_question': None
            }
        
        # Handle thanks
        thanks = ['thanks', 'thank you', 'thx', 'ty']
        if any(t in user_question.lower() for t in thanks):
            return {
                'answer': "You're welcome! 😊 Is there anything else I can help you with?",
                'confidence': 1.0,
                'matched_question': None
            }
        
        # Find best matching FAQ
        best_idx, confidence = self.find_best_match(user_question)
        
        # Set confidence threshold
        CONFIDENCE_THRESHOLD = 0.2
        
        if confidence >= CONFIDENCE_THRESHOLD:
            return {
                'answer': self.answers[best_idx],
                'confidence': round(confidence, 2),
                'matched_question': self.questions[best_idx]
            }
        else:
            return {
                'answer': "I'm sorry, I couldn't find a relevant answer to your question. Please try rephrasing or contact our support team for assistance.",
                'confidence': round(confidence, 2),
                'matched_question': None
            }
    
    def get_all_faqs(self):
        """Return all FAQs for display"""
        return [{'question': q, 'answer': a} for q, a in zip(self.questions, self.answers)]

def chat():

import numpy as np
from music21 import converter, instrument, note, chord, stream
from tensorflow.keras.models import Sequential, load_model
from tensorflow.keras.layers import LSTM, Dense, Dropout, Activation, BatchNormalization
from tensorflow.keras.callbacks import ModelCheckpoint, EarlyStopping
from tensorflow.keras.utils import to_categorical
import os
import pickle
import glob

class MusicGenerator:
    def __init__(self, sequence_length=100):
        self.sequence_length = sequence_length
        self.notes = []
        self.note_to_int = {}
        self.int_to_note = {}
        self.n_vocab = 0
        self.model = None
        
    def get_notes_from_midi(self, midi_folder='midi_files'):
        """Extract notes and chords from MIDI files"""
        notes = []
        
        # Get all MIDI files
        midi_files = glob.glob(os.path.join(midi_folder, '*.mid')) + \
                     glob.glob(os.path.join(midi_folder, '*.midi'))
        
        if not midi_files:
            print(f"No MIDI files found in {midi_folder}")
            print("Creating sample MIDI for demonstration...")
            self._create_sample_midi(midi_folder)
            midi_files = glob.glob(os.path.join(midi_folder, '*.mid'))
        
        print(f"Processing {len(midi_files)} MIDI files...")
        
        for file in midi_files:
            try:
                midi = converter.parse(file)
                print(f"Parsing: {file}")
                
                notes_to_parse = None
                
                # Handle different MIDI structures
                try:
                    # File has instrument parts
                    parts = instrument.partitionByInstrument(midi)
                    notes_to_parse = parts.parts[0].recurse()
                except:
                    # File has notes in flat structure
                    notes_to_parse = midi.flat.notes
                
                for element in notes_to_parse:
                    if isinstance(element, note.Note):
                        notes.append(str(element.pitch))
                    elif isinstance(element, chord.Chord):
                        notes.append('.'.join(str(n) for n in element.normalOrder))
                        
            except Exception as e:
                print(f"Error processing {file}: {e}")
                continue
        
        self.notes = notes
        print(f"Total notes extracted: {len(notes)}")
        return notes
    
    def _create_sample_midi(self, folder):
        """Create a sample MIDI file for demonstration"""
        os.makedirs(folder, exist_ok=True)
        
        # Create a simple melody
        s = stream.Stream()
        
        # C major scale pattern repeated with variations
        patterns = [
            ['C4', 'E4', 'G4', 'C5', 'G4', 'E4', 'C4', 'D4'],
            ['E4', 'G4', 'B4', 'E5', 'B4', 'G4', 'E4', 'F4'],
            ['G4', 'B4', 'D5', 'G5', 'D5', 'B4', 'G4', 'A4'],
            ['C4', 'D4', 'E4', 'F4', 'G4', 'A4', 'B4', 'C5'],
        ]
        
        for _ in range(5):  # Repeat patterns
            for pattern in patterns:
                for p in pattern:
                    n = note.Note(p)
                    n.duration.quarterLength = 0.5
                    s.append(n)
        
        # Save the MIDI file
        s.write('midi', fp=os.path.join(folder, 'sample.mid'))
        print(f"Created sample MIDI file in {folder}")
    
    def prepare_sequences(self):
        """Prepare input sequences for training"""
        # Get unique notes
        unique_notes = sorted(set(self.notes))
        self.n_vocab = len(unique_notes)
        
        print(f"Unique notes/chords: {self.n_vocab}")
        
        # Create mappings
        self.note_to_int = {note: num for num, note in enumerate(unique_notes)}
        self.int_to_note = {num: note for num, note in enumerate(unique_notes)}
        
        # Create input sequences and outputs
        network_input = []
        network_output = []
        
        for i in range(len(self.notes) - self.sequence_length):
            seq_in = self.notes[i:i + self.sequence_length]
            seq_out = self.notes[i + self.sequence_length]
            
            network_input.append([self.note_to_int[char] for char in seq_in])
            network_output.append(self.note_to_int[seq_out])
        
        n_patterns = len(network_input)
        print(f"Total training sequences: {n_patterns}")
        
        # Reshape for LSTM [samples, time steps, features]
        network_input = np.reshape(network_input, (n_patterns, self.sequence_length, 1))
        # Normalize
        network_input = network_input / float(self.n_vocab)
        # One-hot encode output
        network_output = to_categorical(network_output, num_classes=self.n_vocab)
        
        return network_input, network_output
    
    def build_model(self, network_input):
        """Build the LSTM model"""
        model = Sequential([
            LSTM(256, input_shape=(network_input.shape[1], network_input.shape[2]),
                 return_sequences=True),
            Dropout(0.3),
            BatchNormalization(),
            
            LSTM(256, return_sequences=True),
            Dropout(0.3),
            BatchNormalization(),
            
            LSTM(256),
            Dropout(0.3),
            BatchNormalization(),
            
            Dense(256, activation='relu'),
            Dropout(0.3),
            
            Dense(self.n_vocab, activation='softmax')
        ])
        
        model.compile(loss='categorical_crossentropy', optimizer='adam')
        self.model = model
        
        print(model.summary())
        return model
    
    def train(self, network_input, network_output, epochs=100, batch_size=64):
        """Train the model"""
        # Callbacks
        checkpoint = ModelCheckpoint(
            'music_model_weights.keras',
            monitor='loss',
            verbose=1,
            save_best_only=True,
            mode='min'
        )
        
        early_stop = EarlyStopping(
            monitor='loss',
            patience=10,
            verbose=1,
            restore_best_weights=True
        )
        
        history = self.model.fit(
            network_input, 
            network_output,
            epochs=epochs,
            batch_size=batch_size,
            callbacks=[checkpoint, early_stop],
            verbose=1
        )
        
        return history
    
    def generate_music(self, seed_sequence=None, num_notes=500, temperature=1.0):
        """Generate new music using the trained model"""
        if self.model is None:
            raise ValueError("Model not trained or loaded")
        
        # Use random seed if not provided
        if seed_sequence is None:
            start = np.random.randint(0, len(self.notes) - self.sequence_length)
            pattern = [self.note_to_int[n] for n in self.notes[start:start + self.sequence_length]]
        else:
            pattern = seed_sequence
        
        generated_notes = []
        
        print(f"Generating {num_notes} notes...")
        
        for i in range(num_notes):
            # Prepare input
            prediction_input = np.reshape(pattern, (1, len(pattern), 1))
            prediction_input = prediction_input / float(self.n_vocab)
            
            # Get prediction
            prediction = self.model.predict(prediction_input, verbose=0)
            
            # Apply temperature for creativity
            prediction = np.log(prediction + 1e-7) / temperature
            prediction = np.exp(prediction) / np.sum(np.exp(prediction))
            
            # Sample from the distribution
            index = np.random.choice(len(prediction[0]), p=prediction[0])
            
            result = self.int_to_note[index]
            generated_notes.append(result)
            
            # Update pattern for next prediction
            pattern.append(index)
            pattern = pattern[1:]
            
            if (i + 1) % 100 == 0:
                print(f"Generated {i + 1}/{num_notes} notes")
        
        return generated_notes
    
    def notes_to_midi(self, notes, output_file='output/generated_music.mid'):
        """Convert generated notes to MIDI file"""
        os.makedirs(os.path.dirname(output_file), exist_ok=True)
        
        output_notes = []
        offset = 0
        
        for pattern in notes:
            # Pattern is a chord
            if ('.' in pattern) or pattern.isdigit():
                notes_in_chord = pattern.split('.')
                chord_notes = []
                for current_note in notes_in_chord:
                    try:
                        new_note = note.Note(int(current_note))
                        new_note.storedInstrument = instrument.Piano()
                        chord_notes.append(new_note)
                    except:
                        continue
                if chord_notes:
                    new_chord = chord.Chord(chord_notes)
                    new_chord.offset = offset
                    output_notes.append(new_chord)
            # Pattern is a single note
            else:
                try:
                    new_note = note.Note(pattern)
                    new_note.offset = offset
                    new_note.storedInstrument = instrument.Piano()
                    output_notes.append(new_note)
                except:
                    continue
            
            # Increase offset (time between notes)
            offset += 0.5
        
        midi_stream = stream.Stream(output_notes)
        midi_stream.write('midi', fp=output_file)
        
        print(f"MIDI file saved to: {output_file}")
        return output_file
    
    def save_state(self, filepath='music_generator_state.pkl'):
        """Save the generator state"""
        state = {
            'notes': self.notes,
            'note_to_int': self.note_to_int,
            'int_to_note': self.int_to_note,
            'n_vocab': self.n_vocab,
            'sequence_length': self.sequence_length
        }
        with open(filepath, 'wb') as f:
            pickle.dump(state, f)
        print(f"State saved to {filepath}")
    
    def load_state(self, filepath='music_generator_state.pkl'):
        """Load the generator state"""
        with open(filepath, 'rb') as f:
            state = pickle.load(f)
        self.notes = state['notes']
        self.note_to_int = state['note_to_int']
        self.int_to_note = state['int_to_note']
        self.n_vocab = state['n_vocab']
        self.sequence_length = state['sequence_length']
        print(f"State loaded from {filepath}")
    
    def load_trained_model(self, model_path='music_model_weights.keras'):
        """Load a pre-trained model"""
        self.model = load_model(model_path)
        print(f"Model loaded from {model_path}")

2. simple_tracker.py (Custom SORT-like Tracker)

import numpy as np
from scipy.optimize import linear_sum_assignment

class SimpleTracker:
    """
    Simple Object Tracker using IoU matching (SORT-like implementation)
    """
    def __init__(self, max_disappeared=30, min_hits=3):
        self.next_id = 0
        self.tracked_objects = {}  # {id: {'bbox': [...], 'hits': int, 'age': int}}
        self.max_disappeared = max_disappeared
        self.min_hits = min_hits
        
    def compute_iou(self, box1, box2):
        """Calculate Intersection over Union between two boxes"""
        x1 = max(box1[0], box2[0])
        y1 = max(box1[1], box2[1])
        x2 = min(box1[2], box2[2])
        y2 = min(box1[3], box2[3])
        
        intersection = max(0, x2 - x1) * max(0, y2 - y1)
        
        area1 = (box1[2] - box1[0]) * (box1[3] - box1[1])
        area2 = (box2[2] - box2[0]) * (box2[3] - box2[1])
        
        union = area1 + area2 - intersection
        
        return intersection / union if union > 0 else 0
    
    def update(self, detections):
        """
        Update tracker with new detections
        detections: list of [x1, y1, x2, y2, confidence, class_id]
        Returns: list of [x1, y1, x2, y2, track_id, class_id]
        """
        # If no detections, increase age of all tracked objects
        if len(detections) == 0:
            for track_id in list(self.tracked_objects.keys()):
                self.tracked_objects[track_id]['age'] += 1
                if self.tracked_objects[track_id]['age'] > self.max_disappeared:
                    del self.tracked_objects[track_id]
            return []
        
        # If no tracked objects, register all detections
        if len(self.tracked_objects) == 0:
            for det in detections:
                self._register(det)
            return self._get_active_tracks()
        
        # Calculate IoU matrix between tracked objects and detections
        track_ids = list(self.tracked_objects.keys())
        iou_matrix = np.zeros((len(track_ids), len(detections)))
        
        for i, track_id in enumerate(track_ids):
            for j, det in enumerate(detections):
                iou_matrix[i, j] = self.compute_iou(
                    self.tracked_objects[track_id]['bbox'],
                    det[:4]
                )
        
        # Use Hungarian algorithm for optimal assignment
        row_indices, col_indices = linear_sum_assignment(-iou_matrix)
        
        matched_tracks = set()
        matched_detections = set()
        
        for row, col in zip(row_indices, col_indices):
            if iou_matrix[row, col] >= 0.3:  # IoU threshold
                track_id = track_ids[row]
                det = detections[col]
                self.tracked_objects[track_id]['bbox'] = det[:4]
                self.tracked_objects[track_id]['class_id'] = int(det[5])
                self.tracked_objects[track_id]['hits'] += 1
                self.tracked_objects[track_id]['age'] = 0
                matched_tracks.add(row)
                matched_detections.add(col)
        
        # Handle unmatched tracks (disappeared objects)
        for i, track_id in enumerate(track_ids):
            if i not in matched_tracks:
                self.tracked_objects[track_id]['age'] += 1
                if self.tracked_objects[track_id]['age'] > self.max_disappeared:
                    del self.tracked_objects[track_id]
        
        # Register unmatched detections as new tracks
        for j, det in enumerate(detections):
            if j not in matched_detections:
                self._register(det)
        
        return self._get_active_tracks()
    
    def _register(self, detection):
        """Register a new track"""
        self.tracked_objects[self.next_id] = {
            'bbox': detection[:4],
            'class_id': int(detection[5]),
            'hits': 1,
            'age': 0
        }
        self.next_id += 1
    
    def _get_active_tracks(self):
        """Return tracks that have been seen enough times"""
        tracks = []
        for track_id, obj in self.tracked_objects.items():
            if obj['hits'] >= self.min_hits:
                track = list(obj['bbox']) + [track_id, obj['class_id']]
                tracks.append(track)
        return tracks

3. object_tracker.py (Main Object Detection & Tracking)

import cv2
import numpy as np
from ultralytics import YOLO
from simple_tracker import SimpleTracker
import time

class ObjectDetectorTracker:
    def __init__(self, model_name='yolov8n.pt', confidence=0.5):
        """
        Initialize the object detector and tracker
        
        Args:
            model_name: YOLO model to use (yolov8n, yolov8s, yolov8m, yolov8l, yolov8x)
            confidence: Detection confidence threshold
        """
        print(f"Loading YOLO model: {model_name}")
        self.model = YOLO(model_name)
        self.confidence = confidence
        self.tracker = SimpleTracker(max_disappeared=30, min_hits=3)
        
        # Colors for different track IDs (generate unique colors)
        np.random.seed(42)
        self.colors = np.random.randint(0, 255, size=(1000, 3), dtype=np.uint8)
        
        # COCO class names
        self.class_names = self.model.names
        
    def detect_and_track(self, frame):
        """
        Perform detection and tracking on a frame
        
        Returns:
            frame: Annotated frame
            tracks: List of tracked objects [x1, y1, x2, y2, track_id, class_id]
        """
        # Run YOLO detection
        results = self.model(frame, conf=self.confidence, verbose=False)
        
        # Extract detections
        detections = []
        for result in results:
            boxes = result.boxes
            if boxes is not None:
                for box in boxes:
                    x1, y1, x2, y2 = box.xyxy[0].cpu().numpy()
                    conf = box.conf[0].cpu().numpy()
                    cls = box.cls[0].cpu().numpy()
                    detections.append([x1, y1, x2, y2, conf, cls])
        
        # Update tracker
        tracks = self.tracker.update(detections)
        
        return frame, tracks
    
    def draw_tracks(self, frame, tracks):
        """Draw bounding boxes and labels on the frame"""
        for track in tracks:
            x1, y1, x2, y2, track_id, class_id = track
            x1, y1, x2, y2 = int(x1), int(y1), int(x2), int(y2)
            track_id = int(track_id)
            class_id = int(class_id)
            
            # Get color for this track ID
            color = tuple(map(int, self.colors[track_id % 1000]))
            
            # Draw bounding box
            cv2.rectangle(frame, (x1, y1), (x2, y2), color, 2)
            
            # Prepare label
            class_name = self.class_names.get(class_id, f"Class{class_id}")
            label = f"ID:{track_id} {class_name}"
            
            # Draw label background
            (label_width, label_height), baseline = cv2.getTextSize(
                label, cv2.FONT_HERSHEY_SIMPLEX, 0.6, 2
            )
            cv2.rectangle(
                frame, 
                (x1, y1 - label_height - 10), 
                (x1 + label_width, y1), 
                color, 
                -1
            )
            
            # Draw label text
            cv2.putText(
                frame, label, (x1, y1 - 5),
                cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 255, 255), 2
            )
        
        return frame
    
    def process_video(self, source=0, output_path=None, show=True):
        """
        Process video from webcam or file
        
        Args:
            source: 0 for webcam, or path to video file
            output_path: Path to save output video (optional)
            show: Whether to display the output
        """
        # Open video source
        cap = cv2.VideoCapture(source)
        
        if not cap.isOpened():
            print(f"Error: Could not open video source: {source}")
            return
        
        # Get video properties
        fps = int(cap.get(cv2.CAP_PROP_FPS)) or 30
        width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
        height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
        
        print(f"Video: {width}x{height} @ {fps} FPS")
        
        # Setup video writer if saving output
        writer = None
        if output_path:
            fourcc = cv2.VideoWriter_fourcc(*'mp4v')
            writer = cv2.VideoWriter(output_path, fourcc, fps, (width, height))
        
        frame_count = 0
        start_time = time.time()
        
        print("\nPress 'q' to quit, 's' to save screenshot")
        
        while True:
            ret, frame = cap.read()
            if not ret:
                break
            
            frame_count += 1
            
            # Detect and track objects
            frame, tracks = self.detect_and_track(frame)
            
            # Draw tracked objects
            frame = self.draw_tracks(frame, tracks)
            
            # Calculate and display FPS
            elapsed_time = time.time() - start_time
            fps_display = frame_count / elapsed_time if elapsed_time > 0 else 0
            
            # Draw info overlay
            info_text = f"FPS: {fps_display:.1f} | Objects: {len(tracks)}"
            cv2.putText(
                frame, info_text, (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 255, 0), 2
            )
            
            # Show frame
            if show:
                cv2.imshow("Object Detection & Tracking - CodeAlpha", frame)
            
            # Write frame to output video
            if writer:
                writer.write(frame)
            
            # Handle key presses
            key = cv2.waitKey(1) & 0xFF
            if key == ord('q'):
                break
            elif key == ord('s'):
                screenshot_path = f"screenshot_{frame_count}.jpg"
                cv2.imwrite(screenshot_path, frame)
                print(f"Screenshot saved: {screenshot_path}")
        
        # Cleanup
        cap.release()
        if writer:
            writer.release()
        cv2.destroyAllWindows()
        
        print(f"\nProcessed {frame_count} frames in {elapsed_time:.2f} seconds")
        print(f"Average FPS: {fps_display:.2f}")
    
    def process_single_image(self, image_path, output_path=None):
        """Process a single image"""
        frame = cv2.imread(image_path)
        if frame is None:
            print(f"Error: Could not read image: {image_path}")
            return
        
        # Detect (tracking doesn't make sense for single image)
        results = self.model(frame, conf=self.confidence, verbose=False)
        
        # Draw detections
        for result in results:
            boxes = result.boxes
            if boxes is not None:
                for i, box in enumerate(boxes):
                    x1, y1, x2, y2 = map(int, box.xyxy[0].cpu().numpy())
                    conf = float(box.conf[0].cpu().numpy())
                    cls = int(box.cls[0].cpu().numpy())
                    
                    color = tuple(map(int, self.colors[i]))
                    class_name = self.class_names.get(cls, f"Class{cls}")
                    
                    cv2.rectangle(frame, (x1, y1), (x2, y2), color, 2)
                    label = f"{class_name}: {conf:.2f}"
                    cv2.putText(frame, label, (x1, y1-10),
                               cv2.FONT_HERSHEY_SIMPLEX, 0.6, color, 2)
        
        if output_path:
            cv2.imwrite(output_path, frame)
            print(f"Output saved to: {output_path}")
        
        cv2.imshow("Detection Result", frame)
        cv2.waitKey(0)
        cv2.destroyAllWindows()

from object_tracker import ObjectDetectorTracker
import argparse
import os

def main():
    parser = argparse.ArgumentParser(description='Object Detection and Tracking - CodeAlpha')
    parser.add_argument('--source', type=str, default='0',
                       help='Video source: 0 for webcam, or path to video file')
    parser.add_argument('--model', type=str, default='yolov8n.pt',
                       help='YOLO model to use (yolov8n.pt, yolov8s.pt, yolov8m.pt)')
    parser.add_argument('--confidence', type=float, default=0.5,
                       help='Detection confidence threshold (0-1)')
    parser.add_argument('--output', type=str, default=None,
                       help='Path to save output video')
    parser.add_argument('--no-display', action='store_true',
                       help='Disable display (useful for headless systems)')
    
    args = parser.parse_args()
    
    print("=" * 60)
    print("Object Detection and Tracking - CodeAlpha AI Internship")
    print("=" * 60)
    print(f"\nModel: {args.model}")
    print(f"Confidence Threshold: {args.confidence}")
    print(f"Source: {'Webcam' if args.source == '0' else args.source}")
    
    # Initialize detector and tracker
    detector = ObjectDetectorTracker(
        model_name=args.model,
        confidence=args.confidence
    )
    
    # Determine source type
    source = int(args.source) if args.source.isdigit() else args.source
    
    # Check if source is an image
    if isinstance(source, str) and source.lower().endswith(('.jpg', '.jpeg', '.png', '.bmp')):
        output = args.output or 'detected_' + os.path.basename(source)
        detector.process_single_image(source, output)
    else:
        # Process video
        detector.process_video(
            source=source,
            output_path=args.output,
            show=not args.no_display
        )

if __name__ == '__main__':
    main()
